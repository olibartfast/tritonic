cmake_minimum_required(VERSION 3.24)
project(yolo11_seg_dali_plugin LANGUAGES CUDA CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CUDA_STANDARD 20)
set(CMAKE_CUDA_ARCHITECTURES "86" CACHE STRING
    "CUDA architectures to build for, e.g. 75 or \"86;89\"")
set(DALI_ROOT "/dali" CACHE PATH "DALI package root")
add_library(yolo11_seg_dali SHARED yolo11_seg_postprocess.cu yolo11_seg_mask_postprocess.cu)
target_include_directories(yolo11_seg_dali SYSTEM PRIVATE "${DALI_ROOT}/include")
target_compile_definitions(yolo11_seg_dali PRIVATE _GLIBCXX_USE_CXX11_ABI=1)
target_link_directories(yolo11_seg_dali PRIVATE "${DALI_ROOT}")
target_link_libraries(yolo11_seg_dali PRIVATE dali)
set_target_properties(yolo11_seg_dali PROPERTIES OUTPUT_NAME yolo11_seg_dali)
